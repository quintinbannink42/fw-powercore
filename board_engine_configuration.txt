! PowerCore PDM persistent fields (spliced into engine_configuration_s)
! Access as engineConfiguration->pdmChannelTrip[i] / pdmCan*.
! e-fuse SM reads pdmChannelTrip[] (docs/EFUSE.md).
! CAN consume/status: firmware/pdm_can.cpp + docs/CAN.md.
! Channel order: 1-4 HP1-HP4, 5-12 ADIO1-ADIO8.
!
! split_lines does not trim: `bit` lines must start at column 0 (see alphax-silver).

struct pdm_channel_trip_s
float inrushLimitA;Inrush / fast-short current limit (instant trip).;"A", 1, 0, 0, 200, 1
float ocLimitA;Continuous overcurrent limit (after inrush window).;"A", 1, 0, 0, 200, 1
uint16_t inrushWindowMs;Inrush window after turn-on. OC limit is ignored during this window.;"ms", 1, 0, 0, 10000, 0
uint16_t tripTimeMs;Time above OC limit (after window) before trip.;"ms", 1, 0, 0, 10000, 0
uint8_t retryCount;Auto-retries after a trip before latch. 0 = latch immediately.;"", 1, 0, 0, 20, 0
bit latchOnFault,"yes","no"
end_struct

pdm_channel_trip_s[PDM_CHANNEL_COUNT iterate] pdmChannelTrip

bit pdmCanConsumeEnable,"enabled","disabled"
bit pdmCanStatusEnable,"enabled","disabled"
uint16_t pdmCanConsumeBaseId;rusEFI ECU verboseCanBaseAddress to consume (default 0x200).;"id", 1, 0, 0, 2047, 0
uint16_t pdmCanStatusBaseId;PowerCore status TX base ID (default 0x240). DBC: firmware/powercore_pdm.dbc.;"id", 1, 0, 0, 2047, 0
